public class TaskListViewModel : ObservableObject
{
    private readonly ITaskRepository _repository;
    private ObservableCollection<TaskItem> _tasks = new();
    private string _sortBy = "DueDate";

    public ObservableCollection<TaskItem> Tasks
    {
        get => _tasks;
        set => SetProperty(ref _tasks, value);
    }

    public ICommand SortByDateCommand { get; }
    public ICommand SortByPriorityCommand { get; }
    public ICommand AddTaskCommand { get; }

    public TaskListViewModel(ITaskRepository repository)
    {
        _repository = repository;
        SortByDateCommand = new RelayCommand(() => SortTasks("DueDate"));
        SortByPriorityCommand = new RelayCommand(() => SortTasks("Priority"));
    }

    // QuickSort implementation for sorting tasks
    private void SortTasks(string sortBy)
    {
        var taskList = Tasks.ToList();
        QuickSort(taskList, 0, taskList.Count - 1, sortBy);
        Tasks = new ObservableCollection<TaskItem>(taskList);
    }

    private void QuickSort(List<TaskItem> tasks, int low, int high, string sortBy)
    {
        if (low < high)
        {
            int pivotIndex = Partition(tasks, low, high, sortBy);
            QuickSort(tasks, low, pivotIndex - 1, sortBy);
            QuickSort(tasks, pivotIndex + 1, high, sortBy);
        }
    }

    private int Partition(List<TaskItem> tasks, int low, int high, string sortBy)
    {
        var pivot = tasks[high];
        int i = low - 1;

        for (int j = low; j < high; j++)
        {
            bool shouldSwap = sortBy switch
            {
                "DueDate" => tasks[j].DueDate < pivot.DueDate,
                "Priority" => tasks[j].Priority > pivot.Priority,
                _ => false
            };

            if (shouldSwap)
            {
                i++;
                (tasks[i], tasks[j]) = (tasks[j], tasks[i]);
            }
        }

        (tasks[i + 1], tasks[high]) = (tasks[high], tasks[i + 1]);
        return i + 1;
    }
}

Algorithms/ISortingAlgorithm.cs (Strategy Pattern) 
// Strategy interface
public interface ISortingAlgorithm
{
    string Name { get; }
    string TimeComplexity { get; }
    string SpaceComplexity { get; }
    IAsyncEnumerable<SortStep> Sort(int[] array);
}

// Visualization step
public record SortStep(
    int[] Array,
    int CompareIndex1,
    int CompareIndex2,
    bool IsSwap,
    string Description
);

// Bubble Sort implementation
public class BubbleSort : ISortingAlgorithm
{
    public string Name => "Bubble Sort";
    public string TimeComplexity => "O(n²)";
    public string SpaceComplexity => "O(1)";

    public async IAsyncEnumerable<SortStep> Sort(int[] array)
    {
        int n = array.Length;
        for (int i = 0; i < n - 1; i++)
        {
            for (int j = 0; j < n - i - 1; j++)
            {
                // Yield comparison step
                yield return new SortStep(
                    array.ToArray(), j, j + 1, false,
                    $"Comparing {array[j]} and {array[j + 1]}"
                );

                if (array[j] > array[j + 1])
                {
                    (array[j], array[j + 1]) = (array[j + 1], array[j]);

                    // Yield swap step
                    yield return new SortStep(
                        array.ToArray(), j, j + 1, true,
                        $"Swapped! {array[j + 1]} > {array[j]}"
                    );
                }
            }
        }
    }
}

// Quick Sort implementation
public class QuickSort : ISortingAlgorithm
{
    public string Name => "Quick Sort";
    public string TimeComplexity => "O(n log n)";
    public string SpaceComplexity => "O(log n)";

    public async IAsyncEnumerable<SortStep> Sort(int[] array)
    {
        await foreach (var step in QuickSortHelper(array, 0, array.Length - 1))
        {
            yield return step;
        }
    }

    private async IAsyncEnumerable<SortStep> QuickSortHelper(int[] arr, int low, int high)
    {
        if (low < high)
        {
            int pivotIndex = low;
            await foreach (var step in Partition(arr, low, high))
            {
                pivotIndex = step.CompareIndex1; // Final pivot position
                yield return step;
            }

            await foreach (var step in QuickSortHelper(arr, low, pivotIndex - 1))
                yield return step;
            await foreach (var step in QuickSortHelper(arr, pivotIndex + 1, high))
                yield return step;
        }
    }
    // ... Partition implementation
}

    private int _compareIndex2 = -1;
    private bool _isRunning = false;
    private int _delay = 50;
    private int _comparisons = 0;
    private int _swaps = 0;
    private string _currentStep = "";

    private List<ISortingAlgorithm> _algorithms = new()
    {
        new BubbleSort(),
        new QuickSort(),
        new MergeSort(),
        new HeapSort()
    };

    private async Task StartSort()
    {
        _isRunning = true;
        var algorithm = _algorithms.First(a => a.Name == _selectedAlgorithm);

        await foreach (var step in algorithm.Sort(_array))
        {
            _array = step.Array;
            _compareIndex1 = step.CompareIndex1;
            _compareIndex2 = step.CompareIndex2;
            _currentStep = step.Description;

            if (step.IsSwap) _swaps++;
            else _comparisons++;

            StateHasChanged();
            await Task.Delay(_delay);
        }
        _isRunning = false;
    }

    private string GetBarClass(int index)
    {
        if (index == _compareIndex1 || index == _compareIndex2)
            return "comparing";
        return "";
    }

// Using Dictionary as Hash Table for O(1) category lookups
public class CategoryLookup
{
    private readonly Dictionary<string, Category> _categories = new();
    private readonly Dictionary<string, List<Transaction>> _transactionsByCategory = new();

    public void AddCategory(Category category)
    {
        _categories[category.Name.ToLower()] = category;
        _transactionsByCategory[category.Name.ToLower()] = new List<Transaction>();
    }

    public Category? GetCategory(string name)
    {
        // O(1) average lookup
        return _categories.TryGetValue(name.ToLower(), out var category)
            ? category : null;
    }

    public decimal GetTotalByCategory(string categoryName)
    {
        if (_transactionsByCategory.TryGetValue(categoryName.ToLower(), out var transactions))
        {
            return transactions.Sum(t => t.Amount);
        }
        return 0;
    }
}

public class EditSnippetCommand : ICommand
{
    private readonly ISnippetRepository _repository;
    private readonly Snippet _newState;
    private Snippet _previousState;

    public string Description => $"Edit snippet: {_newState.Title}";

    public void Execute()
    {
        _previousState = _repository.GetById(_newState.Id).Clone();
        _repository.Update(_newState);
    }

    public void Undo() => _repository.Update(_previousState);
}

// Command history manager
public class CommandHistory
{
    private readonly Stack<ICommand> _undoStack = new();
    private readonly Stack<ICommand> _redoStack = new();

    public bool CanUndo => _undoStack.Count > 0;
    public bool CanRedo => _redoStack.Count > 0;

    public void Execute(ICommand command)
    {
        command.Execute();
        _undoStack.Push(command);
        _redoStack.Clear(); // Clear redo after new action
    }

    public void Undo()
    {
        if (!CanUndo) return;
        var command = _undoStack.Pop();
        command.Undo();
        _redoStack.Push(command);
    }

    public void Redo()
    {
        if (!CanRedo) return;
        var command = _redoStack.Pop();
        command.Execute();
        _undoStack.Push(command);
    }
}
DataStructures/Trie.cs (Autocomplete) 
// Trie for fast prefix-based autocomplete
public class TrieNode
{
    public Dictionary<char, TrieNode> Children { get; } = new();
    public bool IsEndOfWord { get; set; }
    public Snippet? Snippet { get; set; }
}

public class SnippetTrie
{
    private readonly TrieNode _root = new();

    public void Insert(Snippet snippet)
    {
        var node = _root;
        foreach (var c in snippet.Title.ToLower())
        {
            if (!node.Children.ContainsKey(c))
                node.Children[c] = new TrieNode();
            node = node.Children[c];
        }
        node.IsEndOfWord = true;
        node.Snippet = snippet;
    }

    public List<Snippet> Search(string prefix, int maxResults = 10)
    {
        var results = new List<Snippet>();
        var node = _root;

        // Navigate to prefix node
        foreach (var c in prefix.ToLower())
        {
            if (!node.Children.ContainsKey(c))
                return results; // No matches
            node = node.Children[c];
        }

        // DFS to find all words with this prefix
        CollectSnippets(node, results, maxResults);
        return results;
    }

    private void CollectSnippets(TrieNode node, List<Snippet> results, int max)
    {
        if (results.Count >= max) return;

        if (node.IsEndOfWord && node.Snippet != null)
            results.Add(node.Snippet);

        foreach (var child in node.Children.Values)
            CollectSnippets(child, results, max);
    }
}

public class ChatHub : Hub
{
    private readonly IConnectionManager _connections;
    private readonly IMessageService _messageService;

    public ChatHub(IConnectionManager connections, IMessageService messageService)
    {
        _connections = connections;
        _messageService = messageService;
    }

    public override async Task OnConnectedAsync()
    {
        var userId = Context.User?.Identity?.Name ?? "Anonymous";
        _connections.AddConnection(userId, Context.ConnectionId);

        // Notify all clients about new user (Observer pattern)
        await Clients.All.SendAsync("UserConnected", userId);
        await base.OnConnectedAsync();
    }

    public override async Task OnDisconnectedAsync(Exception? exception)
    {
        var userId = Context.User?.Identity?.Name ?? "Anonymous";
        _connections.RemoveConnection(Context.ConnectionId);

        await Clients.All.SendAsync("UserDisconnected", userId);
        await base.OnDisconnectedAsync(exception);
    }

    public async Task SendMessage(string roomId, string message)
    {
        var userId = Context.User?.Identity?.Name ?? "Anonymous";

        var chatMessage = new ChatMessage
        {
            UserId = userId,
            RoomId = roomId,
            Content = message,
            Timestamp = DateTime.UtcNow
        };

        // Store message
        await _messageService.SaveMessageAsync(chatMessage);

        // Broadcast to room (Observer pattern)
        await Clients.Group(roomId).SendAsync("ReceiveMessage", chatMessage);
    }

    public async Task JoinRoom(string roomId)
    {
        await Groups.AddToGroupAsync(Context.ConnectionId, roomId);

        // Load recent messages (Queue structure)
        var recentMessages = await _messageService.GetRecentMessagesAsync(roomId, 50);
        await Clients.Caller.SendAsync("LoadMessages", recentMessages);
    }
}

// Singleton Connection Manager
public class ConnectionManager : IConnectionManager
{
    private readonly ConcurrentDictionary<string, HashSet<string>> _userConnections = new();

    public void AddConnection(string userId, string connectionId)
    {
        _userConnections.AddOrUpdate(
            userId,
            _ => new HashSet<string> { connectionId },
            (_, connections) => { connections.Add(connectionId); return connections; }
        );
    }

    public IEnumerable<string> GetConnections(string userId)
    {
        return _userConnections.TryGetValue(userId, out var connections)
            ? connections : Enumerable.Empty<string>();
    }

    public IEnumerable<string> GetOnlineUsers() => _userConnections.Keys;
}

using Elsa;
using Elsa.EntityFrameworkCore.Modules.Management;
using Elsa.EntityFrameworkCore.Modules.Runtime;
using FlowMaster.Workflows.Activities;

var builder = WebApplication.CreateBuilder(args);

// Add Elsa services
builder.Services.AddElsa(elsa =>
{
    // Configure management persistence (workflow definitions)
    elsa.UseWorkflowManagement(management =>
    {
        management.UseEntityFrameworkCore(ef =>
            ef.UsePostgreSql(builder.Configuration.GetConnectionString("Default")));
    });

    // Configure runtime persistence (workflow instances)
    elsa.UseWorkflowRuntime(runtime =>
    {
        runtime.UseEntityFrameworkCore(ef =>
            ef.UsePostgreSql(builder.Configuration.GetConnectionString("Default")));
    });

    // Add HTTP activities
    elsa.UseHttp();

    // Add scheduling activities
    elsa.UseScheduling();

    // Add email activities
    elsa.UseEmail(email =>
    {
        email.ConfigureOptions = options =>
        {
            options.Host = builder.Configuration["Smtp:Host"]!;
            options.Port = int.Parse(builder.Configuration["Smtp:Port"]!);
        };
    });

    // Register custom activities
    elsa.AddActivity<ApprovalActivity>();
    elsa.AddActivity<DataTransformActivity>();
    elsa.AddActivity<NotifyDashboardActivity>();

    // Use Elsa API endpoints
    elsa.UseWorkflowsApi();

FlowMaster.Workflows/Activities/ApprovalActivity.cs 
using Elsa.Extensions;
using Elsa.Workflows;
using Elsa.Workflows.Attributes;
using Elsa.Workflows.Models;

namespace FlowMaster.Workflows.Activities;

/// <summary>
/// Custom activity that waits for human approval before proceeding.
/// Demonstrates bookmarks and long-running workflows.
/// </summary>
[Activity("FlowMaster", "Approval", "Waits for human approval")]
public class ApprovalActivity : Activity
{
    [Input(Description = "The user or group to request approval from")]
    public Input<string> Approver { get; set; } = default!;

    [Input(Description = "Message to display in approval request")]
    public Input<string> Message { get; set; } = default!;

    [Input(Description = "Timeout in hours (0 = no timeout)")]
    public Input<int> TimeoutHours { get; set; } = new(24);

    [Output(Description = "Whether the request was approved")]
    public Output<bool> IsApproved { get; set; } = default!;

    [Output(Description = "Comments from approver")]
    public Output<string> Comments { get; set; } = default!;

    protected override async ValueTask ExecuteAsync(ActivityExecutionContext context)
    {
        var approver = Approver.Get(context);
        var message = Message.Get(context);

        // Create a bookmark to suspend workflow until approval
        var bookmarkPayload = new ApprovalBookmarkPayload(
            context.WorkflowExecutionContext.Id,
            approver,
            message
        );

        context.CreateBookmark(bookmarkPayload, ResumeAsync);
    }

    private ValueTask ResumeAsync(ActivityExecutionContext context)
    {
        // Get the approval response from the bookmark data
        var response = context.GetWorkflowInput<ApprovalResponse>();

        IsApproved.Set(context, response?.Approved ?? false);
        Comments.Set(context, response?.Comments ?? string.Empty);

        return ValueTask.CompletedTask;
    }
}

public record ApprovalBookmarkPayload(string WorkflowId, string Approver, string Message);
public record ApprovalResponse(bool Approved, string Comments);

4 Build Code-Based Workflow Definition
Create workflow definitions programmatically using C# (Strategy Pattern for different workflows).
FlowMaster.Workflows/Workflows/OnboardingWorkflow.cs 
using Elsa.Workflows;
using Elsa.Workflows.Activities;
using Elsa.Workflows.Contracts;
using Elsa.Email.Activities;
using FlowMaster.Workflows.Activities;

namespace FlowMaster.Workflows.Workflows;

/// <summary>
/// Employee onboarding workflow using code-based definition.
/// Demonstrates Strategy Pattern - different onboarding flows for different employee types.
/// </summary>
public class OnboardingWorkflow : WorkflowBase
{
    protected override void Build(IWorkflowBuilder builder)
    {
        // Define workflow variables
        var employeeName = builder.WithVariable<string>();
        var employeeEmail = builder.WithVariable<string>();
        var department = builder.WithVariable<string>();
        var isApproved = builder.WithVariable<bool>();

        builder.Name = "Employee Onboarding";
        builder.Description = "Automated employee onboarding process";

        builder.Root = new Sequence
        {
            Activities =
            {
                // Step 1: Notify dashboard of new onboarding
                new NotifyDashboardActivity
                {
                    Title = new("New Onboarding Started"),
                    Message = new(context => $"Onboarding started for {employeeName.Get(context)}"),
                    NotificationType = new("info")
                },

                // Step 2: Send welcome email
                new SendEmail
                {
                    To = new(context => employeeEmail.Get(context)),
                    Subject = new("Welcome to the Team!"),
                    Body = new(context => $@"
                        <h1>Welcome, {employeeName.Get(context)}!</h1>
                        <p>We're excited to have you join the {department.Get(context)} team.</p>
                        <p>Your onboarding process has begun.</p>
                    ")
                },

                // Step 3: HR Approval (suspends workflow)
                new ApprovalActivity
                {
                    Approver = new("HR-Team"),
                    Message = new(context => $"Please approve onboarding for {employeeName.Get(context)}"),
                    TimeoutHours = new(48),
                    IsApproved = new(isApproved)
                },

                // Step 4: Branch based on approval
                new If
                {
                    Condition = new(context => isApproved.Get(context)),
                    Then = new Sequence
                    {
                        Activities =
                        {
                            // Approved flow
                            new Fork
                            {
                                Branches =
                                {
                                    // Parallel: Set up accounts
                                    new SetVariable<string>
                                    {
                                        Variable = employeeName,
                                        Value = new(context => $"{employeeName.Get(context)} (IT Setup Complete)")
                                    },
                                    // Parallel: Order equipment
                                    new NotifyDashboardActivity
                                    {
                                        Title = new("Equipment Order"),
                                        Message = new(context => $"Order equipment for {employeeName.Get(context)}"),
                                        NotificationType = new("info")
                                    }
                                }
                            },
                            new NotifyDashboardActivity
                            {
                                Title = new("Onboarding Complete"),
                                Message = new(context => $"{employeeName.Get(context)} onboarding completed!"),
                                NotificationType = new("success")
                            }
                        }
                    },
                    Else = new Sequence
                    {
                        Activities =
                        {
                            // Rejected flow
                            new SendEmail
                            {
                                To = new(context => employeeEmail.Get(context)),
                                Subject = new("Onboarding Update"),
                                Body = new("There was an issue with your onboarding. HR will contact you.")
                            },
                            new NotifyDashboardActivity
                            {
                                Title = new("Onboarding Issue"),
                                Message = new(context => $"Onboarding rejected for {employeeName.Get(context)}"),
                                NotificationType = new("warning")
                            }
                        }
                    }
                }
            }
        };
    }
}
5 Create SignalR Hub for Real-Time Updates

using Microsoft.AspNetCore.SignalR;

namespace FlowMaster.Api.Hubs;

public class WorkflowHub : Hub
{
    private readonly ILogger<WorkflowHub> _logger;

    public WorkflowHub(ILogger<WorkflowHub> logger)
    {
        _logger = logger;
    }

    public override async Task OnConnectedAsync()
    {
        _logger.LogInformation("Client connected: {ConnectionId}", Context.ConnectionId);
        await Clients.Caller.SendAsync("Connected", new
        {
            ConnectionId = Context.ConnectionId,
            Message = "Connected to FlowMaster Hub"
        });
        await base.OnConnectedAsync();
    }

    public async Task SubscribeToWorkflow(string workflowId)
    {
        await Groups.AddToGroupAsync(Context.ConnectionId, $"workflow-{workflowId}");
        _logger.LogInformation("Client {ConnectionId} subscribed to workflow {WorkflowId}",
            Context.ConnectionId, workflowId);
    }

    public async Task UnsubscribeFromWorkflow(string workflowId)
    {
        await Groups.RemoveFromGroupAsync(Context.ConnectionId, $"workflow-{workflowId}");
    }

    // Method for broadcasting workflow status updates
    public async Task BroadcastWorkflowStatus(string workflowId, object status)
    {
        await Clients.Group($"workflow-{workflowId}")
            .SendAsync("WorkflowStatusUpdate", status);
    }
}

    [ObservableProperty]
    private WorkflowDefinition? _selectedWorkflow;

    public MainViewModel(IApiService apiService, INavigationService navigationService)

        _navigationService = navigationService;
    }

    [RelayCommand]
    private async Task LoadWorkflowsAsync()
    {
        try
        {
            IsLoading = true;
            StatusMessage = "Loading workflows...";

            var workflows = await _apiService.GetWorkflowsAsync();
            Workflows.Clear();
            foreach (var wf in workflows)
            {
                Workflows.Add(wf);
            }

            StatusMessage = $"Loaded {Workflows.Count} workflows";
        }
        catch (Exception ex)
        {
            StatusMessage = $"Error: {ex.Message}";
        }
        finally
        {
            IsLoading = false;
        }
    }

    [RelayCommand]
    private async Task StartWorkflowAsync()
    {
        if (SelectedWorkflow is null) return;

        try
        {
            IsLoading = true;
            StatusMessage = $"Starting {SelectedWorkflow.Name}...";

            var instanceId = await _apiService.StartWorkflowAsync(SelectedWorkflow.Id);
            StatusMessage = $"Workflow started: {instanceId}";
        }
        catch (Exception ex)
        {
            StatusMessage = $"Error: {ex.Message}";
        }
        finally
        {
            IsLoading = false;
        }
    }

    [RelayCommand]
    private void OpenDesigner()
    {
        _navigationService.NavigateTo<WorkflowDesignerPage>(SelectedWorkflow);
    }

    [RelayCommand]
    private void CreateNewWorkflow()
    {
        _navigationService.NavigateTo<WorkflowDesignerPage>(null);
    }
}

@code {
    private HubConnection? hubConnection;
    private bool isConnected => hubConnection?.State == HubConnectionState.Connected;
    private List<WorkflowInstance> activeInstances = new();
    private List<PendingApproval> pendingApprovals = new();
    private List<Notification> notifications = new();
    private int completedToday = 0;

    protected override async Task OnInitializedAsync()
    {
        hubConnection = new HubConnectionBuilder()
            .WithUrl(Navigation.ToAbsoluteUri("/hubs/workflow"))
            .WithAutomaticReconnect()
            .Build();

        // Handle real-time notifications
        hubConnection.On<Notification>("WorkflowNotification", (notification) =>
        {
            notifications.Add(notification);
            InvokeAsync(StateHasChanged);
        });

        // Handle status updates
        hubConnection.On<WorkflowStatusUpdate>("WorkflowStatusUpdate", (update) =>
        {
            if (update.Status == "Completed")
                completedToday++;

            InvokeAsync(StateHasChanged);
        });

        await hubConnection.StartAsync();
        await LoadDataAsync();
    }

    private async Task ApproveAsync(string approvalId, bool approved)
    {
        await hubConnection!.SendAsync("SubmitApproval", approvalId, approved, "");
        pendingApprovals.RemoveAll(a => a.Id == approvalId);
    }

    public async ValueTask DisposeAsync()
    {
        if (hubConnection is not null)
            await hubConnection.DisposeAsync();
    }
}
